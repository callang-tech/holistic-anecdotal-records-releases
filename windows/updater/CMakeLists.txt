cmake_minimum_required(VERSION 3.14)
project(holistic_local_updater LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The local updater requires Windows.")
endif()
add_executable(holistic_local_updater main.cpp)
target_compile_features(holistic_local_updater PRIVATE cxx_std_17)
target_compile_definitions(holistic_local_updater PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(holistic_local_updater PRIVATE shell32 ole32 rstrtmgr)
if(MSVC)
  target_compile_options(holistic_local_updater PRIVATE /W4 /WX /EHsc)
endif()
